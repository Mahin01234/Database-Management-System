-- 1. customers
CREATE TABLE customers (
    customers_id INT PRIMARY KEY,
    customers_name VARCHAR(50) NOT NULL,
    customers_location VARCHAR(100),          
    customers_Phone_Number VARCHAR(20),       
    customers_Email VARCHAR(100),             
    customers_Address VARCHAR(255),        
    customers_City VARCHAR(50),               
    customers_Country VARCHAR(50),   
) ;

-- 2. orders 
CREATE TABLE orders (
    order_id INT PRIMARY KEY,
    customers_id INT,
    order_date DATE,
    total_amount DECIMAL(10, 2),
    order_status VARCHAR(20) DEFAULT 'Pending',
    FOREIGN KEY (customers_id) REFERENCES customers(customers_id)
);
